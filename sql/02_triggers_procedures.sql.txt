-- TRIGGER E FUNÇÃO DE AUDITORIA AUTOMÁTICA (US08)
CREATE OR REPLACE FUNCTION trg_fn_auditoria_usuarios()
RETURNS TRIGGER AS $$
DECLARE
    v_acao VARCHAR(50);
    v_usuario_id INT;
BEGIN
    IF (TG_OP = 'INSERT') THEN
        v_acao := 'INCLUSAO_USUARIO';
        v_usuario_id := NEW.id;
    ELSIF (TG_OP = 'UPDATE') THEN
        v_acao := 'ATUALIZACAO_USUARIO';
        v_usuario_id := NEW.id;
    ELSIF (TG_OP = 'DELETE') THEN
        v_acao := 'EXCLUSAO_USUARIO';
        v_usuario_id := OLD.id;
    END IF;

    INSERT INTO log_acesso (usuario_id, acao_realizada, endereco_ip, data_hora)
    VALUES (v_usuario_id, v_acao, COALESCE(inet_client_addr()::text, '127.0.0.1'), CURRENT_TIMESTAMP);

    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_auditoria_usuarios_changes ON usuario;

CREATE TRIGGER trg_auditoria_usuarios_changes
AFTER INSERT OR UPDATE OR DELETE ON usuario
FOR EACH ROW EXECUTE FUNCTION trg_fn_auditoria_usuarios();


-- FUNCTION PARA CÁLCULO DO PRAZO COM BASE NO SLA (US05 / US09)
CREATE OR REPLACE FUNCTION fn_calcular_prazo_sla(
    p_servico_id INT,
    p_data_abertura TIMESTAMP
)
RETURNS TIMESTAMP
AS $$
DECLARE
    v_horas_sla INT;
BEGIN
    SELECT COALESCE(tempo_sla_horas, 24) INTO v_horas_sla
    FROM servico
    WHERE id = p_servico_id;

    RETURN p_data_abertura + (v_horas_sla || ' hours')::INTERVAL;
END;
$$ LANGUAGE plpgsql;


-- STORED PROCEDURE DE ATRIBUIÇÃO DE SOLICITAÇÃO (US09)
CREATE OR REPLACE PROCEDURE sp_atribuir_solicitacao(
    p_solicitacao_id INT,
    p_equipe_id INT,
    p_colaborador_id INT,
    p_admin_id INT,
    OUT p_resultado VARCHAR(200)
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_status_atual VARCHAR(50);
BEGIN
    SELECT status INTO v_status_atual 
    FROM solicitacao 
    WHERE id = p_solicitacao_id;

    IF v_status_atual IS NULL THEN
        p_resultado := 'Erro: Solicitação não encontrada.';
        RETURN;
    END IF;

    IF v_status_atual NOT IN ('Aberta', 'Em triagem') THEN
        p_resultado := 'Erro: Status inválido para atribuição.';
        RETURN;
    END IF;

    UPDATE solicitacao 
    SET status = 'Em atendimento'
    WHERE id = p_solicitacao_id;

    INSERT INTO tarefa (solicitacao_id, equipe_id, colaborador_id, status_tarefa, descricao, data_atribuicao, versao_concorrencia)
    VALUES (p_solicitacao_id, p_equipe_id, p_colaborador_id, 'Pendente', 'Atendimento de solicitação triada pelo painel administrativo.', CURRENT_TIMESTAMP, 1);

    INSERT INTO log_acesso (usuario_id, acao_realizada, endereco_ip, data_hora)
    VALUES (p_admin_id, 'ATRIBUICAO_SOLICITACAO_' || p_solicitacao_id, '127.0.0.1', CURRENT_TIMESTAMP);

    p_resultado := 'Sucesso: Solicitação triada e tarefa atribuída com êxito.';
END;
$$;


-- CONTROLE DE CONCORRÊNCIA OTIMISTA PARA O MOBILE (US10)
CREATE OR REPLACE PROCEDURE sp_atualizar_tarefa_mobile(
    p_tarefa_id INT,
    p_colaborador_id INT,
    p_novo_status VARCHAR(50),
    p_versao_atual INT,
    OUT p_sucesso BOOLEAN
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_linhas INT;
BEGIN
    UPDATE tarefa
    SET status_tarefa = p_novo_status,
        versao_concorrencia = versao_concorrencia + 1
    WHERE id = p_tarefa_id 
      AND colaborador_id = p_colaborador_id 
      AND versao_concorrencia = p_versao_atual;

    GET DIAGNOSTICS v_linhas = ROW_COUNT;

    IF v_linhas > 0 THEN
        p_sucesso := TRUE;
    ELSE
        p_sucesso := FALSE;
    END IF;
END;
$$;


-- PROCEDIMENTO DE ANONIMIZAÇÃO DE DADOS PARA A LGPD (US15)
CREATE OR REPLACE PROCEDURE sp_anonimizar_cliente_lgpd(
    p_cliente_id INT,
    p_admin_id INT
)
LANGUAGE plpgsql
AS $$
BEGIN
    UPDATE cliente
    SET razao_social = 'ANONIMIZADO LGPD',
        nome_fantasia = 'ANONIMIZADO LGPD',
        cnpj = '00.000.000/0000-00',
        email_contato = 'privacidade@anonimizado.local',
        telefone = '(00) 0000-0000',
        lgpd_anonimizado = TRUE
    WHERE id = p_cliente_id;

    INSERT INTO log_acesso (usuario_id, acao_realizada, endereco_ip, data_hora)
    VALUES (p_admin_id, 'ANONIMIZACAO_LGPD_CLIENTE_' || p_cliente_id, '127.0.0.1', CURRENT_TIMESTAMP);
END;
$$;
